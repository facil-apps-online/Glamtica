-- Vista para mostrar productos con información de categoría
CREATE OR REPLACE VIEW product_details WITH (security_invoker=on) AS
SELECT 
  p.id,
  p.name,
  p.description,
  p.price,
  p.stock_quantity,
  p.is_active,
  c.id AS category_id,
  c.name AS category_name
FROM 
  products p
LEFT JOIN 
  categories c ON p.category_id = c.id;

-- Vista para mostrar órdenes con información de cliente
CREATE OR REPLACE VIEW order_summary WITH (security_invoker=on) AS
SELECT 
  o.id AS order_id,
  o.order_date,
  o.status,
  o.total_amount,
  c.id AS customer_id,
  c.first_name || ' ' || c.last_name AS customer_name,
  c.email AS customer_email,
  COUNT(oi.id) AS total_items,
  o.shipping_address,
  o.shipping_city,
  o.shipping_country
FROM 
  orders o
JOIN 
  customers c ON o.customer_id = c.id
LEFT JOIN 
  order_items oi ON o.id = oi.order_id
GROUP BY 
  o.id, c.id;

-- Vista para detalles completos de órdenes
CREATE OR REPLACE VIEW order_details WITH (security_invoker=on) AS
SELECT 
  o.id AS order_id,
  o.order_date,
  o.status,
  c.id AS customer_id,
  c.first_name || ' ' || c.last_name AS customer_name,
  p.id AS product_id,
  p.name AS product_name,
  oi.quantity,
  oi.unit_price,
  oi.subtotal,
  cat.name AS category_name
FROM 
  orders o
JOIN 
  customers c ON o.customer_id = c.id
JOIN 
  order_items oi ON o.id = oi.order_id
JOIN 
  products p ON oi.product_id = p.id
LEFT JOIN 
  categories cat ON p.category_id = cat.id;
