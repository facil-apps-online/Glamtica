-- Función para actualizar el timestamp de actualización
CREATE OR REPLACE FUNCTION update_updated_at()
RETURNS TRIGGER AS $$
BEGIN
  NEW.updated_at = now();
  RETURN NEW;
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

-- Triggers para actualizar el timestamp en todas las tablas
CREATE TRIGGER update_categories_updated_at
BEFORE UPDATE ON categories
FOR EACH ROW EXECUTE FUNCTION update_updated_at();

CREATE TRIGGER update_products_updated_at
BEFORE UPDATE ON products
FOR EACH ROW EXECUTE FUNCTION update_updated_at();

CREATE TRIGGER update_customers_updated_at
BEFORE UPDATE ON customers
FOR EACH ROW EXECUTE FUNCTION update_updated_at();

CREATE TRIGGER update_profiles_updated_at
BEFORE UPDATE ON profiles
FOR EACH ROW EXECUTE FUNCTION update_updated_at();

CREATE TRIGGER update_orders_updated_at
BEFORE UPDATE ON orders
FOR EACH ROW EXECUTE FUNCTION update_updated_at();

CREATE TRIGGER update_order_items_updated_at
BEFORE UPDATE ON order_items
FOR EACH ROW EXECUTE FUNCTION update_updated_at();

-- Función para actualizar el stock de productos al crear/modificar un item de orden
CREATE OR REPLACE FUNCTION update_product_stock()
RETURNS TRIGGER AS $$
DECLARE
  old_quantity INTEGER := 0;
BEGIN
  -- Si es una actualización, restaurar el stock anterior
  IF TG_OP = 'UPDATE' THEN
    old_quantity := OLD.quantity;
    -- Restaurar el stock anterior
    UPDATE products 
    SET stock_quantity = stock_quantity + old_quantity
    WHERE id = OLD.product_id;
  END IF;
  
  -- Verificar si hay suficiente stock
  IF TG_OP IN ('INSERT', 'UPDATE') THEN
    IF (SELECT stock_quantity FROM products WHERE id = NEW.product_id) < NEW.quantity THEN
      RAISE EXCEPTION 'No hay suficiente stock para el producto con ID %', NEW.product_id;
    END IF;
    
    -- Reducir el stock
    UPDATE products 
    SET stock_quantity = stock_quantity - NEW.quantity
    WHERE id = NEW.product_id;
  END IF;
  
  -- Si es una eliminación, restaurar el stock
  IF TG_OP = 'DELETE' THEN
    UPDATE products 
    SET stock_quantity = stock_quantity + OLD.quantity
    WHERE id = OLD.product_id;
  END IF;
  
  RETURN NULL;
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

-- Triggers para actualizar el stock de productos
CREATE TRIGGER update_stock_on_order_item_insert
AFTER INSERT ON order_items
FOR EACH ROW EXECUTE FUNCTION update_product_stock();

CREATE TRIGGER update_stock_on_order_item_update
AFTER UPDATE ON order_items
FOR EACH ROW EXECUTE FUNCTION update_product_stock();

CREATE TRIGGER update_stock_on_order_item_delete
AFTER DELETE ON order_items
FOR EACH ROW EXECUTE FUNCTION update_product_stock();

-- Función para calcular y actualizar el total de una orden
CREATE OR REPLACE FUNCTION calculate_order_total(order_id_param bigint)
RETURNS numeric AS $$
DECLARE
  total numeric(12,2);
BEGIN
  SELECT COALESCE(SUM(subtotal), 0) INTO total
  FROM order_items
  WHERE order_id = order_id_param;
  
  UPDATE orders
  SET total_amount = total
  WHERE id = order_id_param;
  
  RETURN total;
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

-- Función trigger para actualizar el total de la orden cuando se modifican los items
CREATE OR REPLACE FUNCTION update_order_total()
RETURNS TRIGGER AS $$
BEGIN
  IF TG_OP = 'INSERT' THEN
    PERFORM calculate_order_total(NEW.order_id);
  ELSIF TG_OP = 'UPDATE' THEN
    PERFORM calculate_order_total(NEW.order_id);
    IF OLD.order_id <> NEW.order_id THEN
      PERFORM calculate_order_total(OLD.order_id);
    END IF;
  ELSIF TG_OP = 'DELETE' THEN
    PERFORM calculate_order_total(OLD.order_id);
  END IF;
  RETURN NULL;
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

-- Trigger para actualizar el total de la orden
CREATE TRIGGER update_order_total_on_item_change
AFTER INSERT OR UPDATE OR DELETE ON order_items
FOR EACH ROW EXECUTE FUNCTION update_order_total();

-- Función para crear automáticamente un perfil cuando se registra un nuevo usuario
CREATE OR REPLACE FUNCTION handle_new_user()
RETURNS TRIGGER AS $$
BEGIN
  INSERT INTO public.profiles (id, first_name, last_name, avatar_url, role)
  VALUES (
    NEW.id,
    NEW.raw_user_meta_data->>'first_name',
    NEW.raw_user_meta_data->>'last_name',
    NEW.raw_user_meta_data->>'avatar_url',
    'customer'
  );
  RETURN NEW;
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

-- Trigger para ejecutar la función cuando se crea un nuevo usuario
CREATE TRIGGER on_auth_user_created
  AFTER INSERT ON auth.users
  FOR EACH ROW EXECUTE FUNCTION handle_new_user();

-- Función para buscar productos por nombre o descripción
CREATE OR REPLACE FUNCTION search_products(search_term text)
RETURNS TABLE (
  id bigint,
  name text,
  description text,
  price numeric,
  stock_quantity integer,
  category_name text
) AS $$
BEGIN
  RETURN QUERY
  SELECT 
    p.id,
    p.name,
    p.description,
    p.price,
    p.stock_quantity,
    c.name as category_name
  FROM 
    products p
  LEFT JOIN 
    categories c ON p.category_id = c.id
  WHERE 
    p.is_active = true AND
    (p.name ILIKE '%' || search_term || '%' OR p.description ILIKE '%' || search_term || '%');
END;
$$ LANGUAGE plpgsql SECURITY INVOKER;
